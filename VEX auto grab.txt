local function saveConfig()
	pcall(function()
		if not writefile then return end
		local data = {
			autoSteal = AutoStealEnabled,
			stealMode = StealMode,
			stealRadius = StealRadius,
			selectedBG = selectedBG,
			stealBarScale = StealBarScale,
			stealBarOrientation = StealBarOrientation,
			stealBarColor = StealBarColorName,
			locked = stealBarLocked,
		}
		if stealBarFrameRef then
			local p = stealBarFrameRef.Position
			data.barPosXS, data.barPosX, data.barPosYS, data.barPosY = p.X.Scale, p.X.Offset, p.Y.Scale, p.Y.Offset
		end
		if mainFrameRef then
			local p = mainFrameRef.Position
			data.mainPosXS, data.mainPosX, data.mainPosYS, data.mainPosY = p.X.Scale, p.X.Offset, p.Y.Scale, p.Y.Offset
		end
		writefile(CONFIG_KEY, HttpService:JSONEncode(data))
	end)
end

local function loadConfig()
	local ok, raw = pcall(function()
		if readfile then return readfile(CONFIG_KEY) end
	end)
	if ok and raw then
		local ok2, d = pcall(function() return HttpService:JSONDecode(raw) end)
		if ok2 and d then return d end
	end
	return nil
end

local function isMyPlotByName(plotName)
	local plots = workspace:FindFirstChild("Plots"); if not plots then return false end
	local plot = plots:FindFirstChild(plotName); if not plot then return false end
	local sign = plot:FindFirstChild("PlotSign")
	if sign then
		local yb = sign:FindFirstChild("YourBase")
		if yb and yb:IsA("BillboardGui") then return yb.Enabled == true end
	end
	return false
end

local function getPromptPosition(prompt)
	local att = prompt:FindFirstChild("Attachment")
	if att then return att.WorldPosition or att.Position end
	local p = prompt.Parent
	while p and p ~= workspace do
		if p.Name == "Spawn" and p:IsA("BasePart") then return p.Position end
		p = p.Parent
	end
	if prompt.Parent and prompt.Parent:IsA("BasePart") then return prompt.Parent.Position end
	return nil
end

local function findNearestPrompt()
	local hrp = getHRP(); if not hrp then return nil end
	local plots = workspace:FindFirstChild("Plots"); if not plots then return nil end
	local nearest, dist = nil, math.huge
	for _, plot in ipairs(plots:GetChildren()) do
		if isMyPlotByName(plot.Name) then continue end
		local pods = plot:FindFirstChild("AnimalPodiums"); if not pods then continue end
		for _, pod in ipairs(pods:GetChildren()) do
			local base = pod:FindFirstChild("Base"); if not base then continue end
			local spawn = base:FindFirstChild("Spawn"); if not spawn then continue end
			local d = (spawn.Position - hrp.Position).Magnitude
			if d <= StealRadius and d < dist then
				local att = spawn:FindFirstChild("PromptAttachment")
				if att then
					for _, p in ipairs(att:GetChildren()) do
						if p:IsA("ProximityPrompt") and p.ActionText and p.ActionText:find("Steal") then
							nearest, dist = p, d
						end
					end
				end
			end
		end
	end
	return nearest
end

local function executeSteal(prompt)
	if isStealing or not AutoStealEnabled then return end
	if not StealData[prompt] then
		StealData[prompt] = { hold = {}, trigger = {}, ready = true }
		if getconnections then
			for _, c in ipairs(getconnections(prompt.PromptButtonHoldBegan)) do
				if c.Function then table.insert(StealData[prompt].hold, c.Function) end
			end
			for _, c in ipairs(getconnections(prompt.Triggered)) do
				if c.Function then table.insert(StealData[prompt].trigger, c.Function) end
			end
		end
	end
	local data = StealData[prompt]
	if not data.ready then return end
	data.ready = false
	isStealing = true
	stealVisualPct = 0
	task.spawn(function()
		for _, f in ipairs(data.hold) do pcall(f) end
	end)
	local cfg = STEAL_MODE_CFG[StealMode] or STEAL_MODE_CFG[1]
	local threshold = cfg.threshold
	local nearDist = cfg.nearDist
	local totalTime = tonumber(StealDuration) or 1.3
	local timeToThreshold = totalTime * threshold
	local timeAfterThreshold = totalTime - timeToThreshold
	local startTime = tick()
	while tick() - startTime < timeToThreshold do
		if not AutoStealEnabled then
			isStealing = false; data.ready = true; stealVisualPct = 0; return
		end
		stealVisualPct = math.clamp((tick() - startTime) / totalTime, 0, threshold)
		task.wait()
	end
	stealVisualPct = threshold
	local hrp = getHRP()
	local stillNear = false
	if hrp then
		local targetPos = getPromptPosition(prompt)
		if targetPos and (targetPos - hrp.Position).Magnitude <= nearDist then
			stillNear = true
		end
	end
	if not stillNear then
		local holdStart = tick()
		while tick() - holdStart < 4 do
			if not AutoStealEnabled then
				isStealing = false; data.ready = true; stealVisualPct = 0; return
			end
			stealVisualPct = threshold
			local hrp2 = getHRP()
			if hrp2 then
				local tp = getPromptPosition(prompt)
				if tp and (tp - hrp2.Position).Magnitude <= nearDist then
					stillNear = true; break
				end
			end
			task.wait()
		end
		if not stillNear then
			isStealing = false; data.ready = true; stealVisualPct = 0; return
		end
	end
	local resumeTime = tick()
	while tick() - resumeTime < timeAfterThreshold do
		if not AutoStealEnabled then
			isStealing = false; data.ready = true; stealVisualPct = 0; return
		end
		local fin = (tick() - resumeTime) / math.max(timeAfterThreshold, 0.01)
		stealVisualPct = threshold + fin * (1 - threshold)
		task.wait()
	end
	stealVisualPct = 1
	for _, f in ipairs(data.trigger) do pcall(f) end
	task.wait(0.05)
	data.ready = true
	isStealing = false
	stealVisualPct = 0
end

local function startAutoSteal()
	if stealConn then return end
	stealConn = RunService.Heartbeat:Connect(function()
		if isStealing or not AutoStealEnabled then return end
		local ok, prompt = pcall(findNearestPrompt)
		if ok and prompt then pcall(executeSteal, prompt) end
	end)
end

local function stopAutoSteal()
	if stealConn then stealConn:Disconnect(); stealConn = nil end
	isStealing = false
	stealVisualPct = 0
end

		refreshPctButtons()
	end
	if cfg.stealRadius then
		StealRadius = math.clamp(math.floor(tonumber(cfg.stealRadius) or 60), 1, 500)
		radiusValBox.Text = tostring(StealRadius)
	end
	if cfg.selectedBG then
		local idx = tonumber(cfg.selectedBG)
		if idx and BG_IDS[idx] then
			selectedBG = idx
			bgImage.Image = BG_IDS[idx]
			for j, stroke in ipairs(bgThumbs) do
				stroke.Color = j == idx and Color3.fromRGB(180, 255, 180) or Color3.fromRGB(55, 55, 55)
			end
		end
	end
	if cfg.stealBarScale then
		StealBarScale = math.clamp(math.floor(tonumber(cfg.stealBarScale) or 100), 50, 250)
		sizeValBox.Text = StealBarScale .. "%"
	end
	if cfg.stealBarOrientation then
		StealBarOrientation = cfg.stealBarOrientation == "Vertical" and "Vertical" or "Normal"
		refreshStyleButtons()
	end
	if cfg.stealBarColor then
		for _, c in ipairs(BAR_COLORS) do
			if c.name == cfg.stealBarColor then StealBarColorName = c.name end
		end
		refreshColorSwatches()
		applyBarColor()
	end
	if cfg.barPosX and cfg.barPosY then
		stealBarFrame.Position = UDim2.new(cfg.barPosXS or 0.5, cfg.barPosX, cfg.barPosYS or 0.86, cfg.barPosY)
	end
	if cfg.mainPosX and cfg.mainPosY then
		main.Position = UDim2.new(cfg.mainPosXS or 0.5, cfg.mainPosX, cfg.mainPosYS or 0.5, cfg.mainPosY)
	end
	if cfg.locked ~= nil then
		stealBarLocked = cfg.locked and true or false
		applyLockVisual()
	end
	if cfg.autoSteal then
		AutoStealEnabled = true
		startAutoSteal()
		updateToggleVisual(true)
	end
	updateStealBarLayout()
end)
